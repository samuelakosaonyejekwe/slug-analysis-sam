# Matplotlib is needed only for chart output (the solver runs headless without it).
numpy>=1.24,<3

#                       unsmoothed field). Neither the solver nor the tests need it.
matplotlib>=3.6,<4   # only used for PNG charts; --no-plots needs no Matplotlib
